1) Design a SQL query to identify customers who have made purchases in three consecutive months but then stopped for at least two months, using a sample sales table.

sales_table
user_id, order_date

with cte as (
    select 
        user_id,
        MONTH(order_date) as order_month,
    from sales_table
    group by user_id, MONTH(order_date)
),
with cte2 as (
    select
        *
        LAG(order_month, 1) over(partiton by user_id) as perivous_month,
        LAG(order_month, 2) over(partiton by user_id) as perivous_month_2,
    from cte
),
with cte3 as (
    select 
        *,
        case
            when order_month - 1 =  perivous_month AND order_month - 2 =  perivous_month_2 then 'YES'
            ELSE 'NO' 
        END as result
    from cte2
)
select user_id from cte3 where result = 'YES'
